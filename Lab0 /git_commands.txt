git clone скачать репозиторий с Гитхаба.
git status посмотреть, что поменялось.
git add добавить файлы для сохранения.
git commit сохранить с комментарием.
git push отправить на Гитхаб.
git pull забрать чужие изменения.
git branch ветки.
git checkout переключиться между ветками.
git merge слить ветки.
git log история.
